CREATE TABLE Usuarios
(
    id_usuario INT PRIMARY KEY IDENTITY(1,1),

    username VARCHAR(50) NOT NULL UNIQUE,

    password VARCHAR(255) NOT NULL
)
GO

CREATE TABLE Bitacora
(
    id_bitacora INT PRIMARY KEY IDENTITY(1,1),

    id_usuario INT NOT NULL,

    mensaje VARCHAR(255) NOT NULL,

    fecha DATETIME NOT NULL,

    estado VARCHAR(50),

    CONSTRAINT FK_Bitacora_Usuario
    FOREIGN KEY (id_usuario)
    REFERENCES Usuarios(id_usuario)
)
GO

CREATE PROCEDURE BuscarUsuario
(
    @nombre VARCHAR(50)
)
AS
BEGIN

    SELECT
        id_usuario,
        username,
        password,
	email,
	telefono
    FROM Usuarios
    WHERE username = @nombre

END
GO

CREATE PROCEDURE InsertarBitacora
(
    @idUsuario INT,
    @mensaje VARCHAR(255),
    @fecha DATETIME,
    @estado VARCHAR(50)	
)
AS
BEGIN

    INSERT INTO Bitacora
    (
        id_usuario,
        mensaje,
        fecha,
        estado
    )
    VALUES
    (
        @idUsuario,
        @mensaje,
        @fecha
        @estado
    )

END
GO

CREATE PROCEDURE GuardarUsuario
(
    @username VARCHAR(50),
    @password VARCHAR(255)
)
AS
BEGIN

    INSERT INTO Usuarios
    (
        username,
        password
    )
    VALUES
    (
        @username,
        @password
    )

END
GO

CREATE PROCEDURE ObtenerBitacora
(
    @fechaDesde DATETIME = NULL,
    @fechaHasta DATETIME = NULL
)
AS
BEGIN

    SELECT
        B.id_bitacora,
        U.username,
        B.mensaje,
        B.fecha,
	B.estado
    FROM Bitacora B
    INNER JOIN Usuarios U
        ON B.id_usuario = U.id_usuario
    WHERE
        (@fechaDesde IS NULL
            OR B.fecha >= @fechaDesde)
    AND
        (@fechaHasta IS NULL
            OR B.fecha <= @fechaHasta)
    ORDER BY B.fecha DESC

END
GO

CREATE PROCEDURE ActualizarUsuario
(
    @Id INT,
    @Username VARCHAR(100),
    @PasswordHash VARCHAR(200),
    @Email VARCHAR(100),
    @Telefono VARCHAR(30)
)
AS
BEGIN

    UPDATE Usuarios
    SET
        username = @Username,
        password = @PasswordHash,
	Email=@Email,
	Telefono=@Telefono
    WHERE id_usuario = @Id;

END
GO

CREATE TABLE AuditoriaUsuarios
(
    IdAuditoria INT IDENTITY(1,1) PRIMARY KEY,

    UsuarioId INT NOT NULL,

    Username VARCHAR(100) NOT NULL,

    PasswordHash VARCHAR(255) NOT NULL,

    Fecha DATETIME NOT NULL,

    ModificadoPor VARCHAR(100) NOT NULL,

    Accion VARCHAR(50) NOT NULL,

    CONSTRAINT FK_AuditoriaUsuarios_Usuarios
        FOREIGN KEY (UsuarioId)
        REFERENCES Usuarios(id_usuario)
);
GO

CREATE PROCEDURE [dbo].[Auditoria_ObtenerHistorial]
    @UsuarioId INT
AS
BEGIN
    SELECT * FROM AuditoriaUsuarios
    WHERE UsuarioId = @UsuarioId
    ORDER BY Fecha DESC
END
GO

CREATE PROCEDURE [dbo].[Auditoria_ObtenerPorId]
    @IdAuditoria INT
AS
BEGIN
    SELECT * FROM AuditoriaUsuarios WHERE IdAuditoria = @IdAuditoria
END
GO

CREATE PROCEDURE [dbo].[Auditoria_Insertar]
    @UsuarioId INT,
    @EmailAnterior VARCHAR(100),
    @EmailActual VARCHAR(100),
    @Fecha DATETIME,
    @ModificadoPor VARCHAR(50),
    @Accion VARCHAR(50)
AS
BEGIN
    INSERT INTO AuditoriaUsuarios 
    (UsuarioId, EmailAnterior, EmailActual, Fecha, ModificadoPor, Accion)
    VALUES (@UsuarioId, @EmailAnterior, , @EmailActual, , @Fecha, @ModificadoPor, @Accion)
END
GO

/*=====================================================
  TABLA: Permiso
=====================================================*/
CREATE TABLE Permiso
(
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Nombre NVARCHAR(100) NOT NULL,
    Tipo NVARCHAR(20) NOT NULL
);
GO

/*=====================================================
  TABLA: Permiso_Permiso
=====================================================*/
CREATE TABLE Permiso_Permiso
(
    IdPermisoPadre INT NOT NULL,
    IdPermisoHijo INT NOT NULL,

    CONSTRAINT PK_Permiso_Permiso
        PRIMARY KEY (IdPermisoPadre, IdPermisoHijo),

    CONSTRAINT FK_PP_Padre
        FOREIGN KEY (IdPermisoPadre)
        REFERENCES Permiso(Id),

    CONSTRAINT FK_PP_Hijo
        FOREIGN KEY (IdPermisoHijo)
        REFERENCES Permiso(Id)
);
GO

/*=====================================================
  TABLA: Usuario_Permiso
=====================================================*/
CREATE TABLE Usuario_Permiso
(
    IdUsuario INT NOT NULL,
    IdPermiso INT NOT NULL,

    CONSTRAINT PK_Usuario_Permiso
        PRIMARY KEY (IdUsuario, IdPermiso),

    CONSTRAINT FK_UP_Usuario
        FOREIGN KEY (IdUsuario)
        REFERENCES Usuario(id_usuario),

    CONSTRAINT FK_UP_Permiso
        FOREIGN KEY (IdPermiso)
        REFERENCES Permiso(Id)
);
GO
