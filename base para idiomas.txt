CREATE PROCEDURE ObtenerTraducciones
@IdIdioma INT
AS
BEGIN

SELECT

E.Clave,

T.Texto

FROM Traduccion T

INNER JOIN Etiqueta E
ON T.IdEtiqueta = E.IdEtiqueta

WHERE T.IdIdioma = @IdIdioma

END
GO

CREATE PROC ObtenerIdiomas
AS
BEGIN

SELECT *

FROM Idioma

ORDER BY Nombre

END

ALTER PROCEDURE [dbo].[BuscarUsuario]
(
    @nombre VARCHAR(50)
)
AS
BEGIN

    SELECT
        id_usuario,
        username,
        password,
		email,
		telefono,
		IdIdioma
    FROM Usuarios
    WHERE username = @nombre

END
GO

  CREATE PROCEDURE Usuario_ActualizarIdioma
(
    @IdUsuario INT,
    @IdIdioma INT
)
AS
BEGIN

    UPDATE Usuarios
       SET IdIdioma=@IdIdioma
     WHERE id_usuario=@IdUsuario

END
