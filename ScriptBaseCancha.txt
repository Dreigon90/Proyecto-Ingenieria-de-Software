CREATE TABLE Usuarios
(
    id_usuario INT PRIMARY KEY IDENTITY(1,1),

    username VARCHAR(50) NOT NULL UNIQUE,

    password VARCHAR(255) NOT NULL
)
GO

CREATE TABLE Bitacora
(
    id_bitacora INT PRIMARY KEY IDENTITY(1,1),

    id_usuario INT NOT NULL,

    mensaje VARCHAR(255) NOT NULL,

    fecha DATETIME NOT NULL,

    estado VARCHAR(50),

    CONSTRAINT FK_Bitacora_Usuario
    FOREIGN KEY (id_usuario)
    REFERENCES Usuarios(id_usuario)
)
GO

CREATE PROCEDURE BuscarUsuario
(
    @nombre VARCHAR(50)
)
AS
BEGIN

    SELECT
        id_usuario,
        username,
        password,
	email,
	telefono
    FROM Usuarios
    WHERE username = @nombre

END
GO

CREATE PROCEDURE InsertarBitacora
(
    @idUsuario INT,
    @mensaje VARCHAR(255),
    @fecha DATETIME,
    @estado VARCHAR(50)	
)
AS
BEGIN

    INSERT INTO Bitacora
    (
        id_usuario,
        mensaje,
        fecha,
        estado
    )
    VALUES
    (
        @idUsuario,
        @mensaje,
        @fecha
        @estado
    )

END
GO

CREATE PROCEDURE GuardarUsuario
(
    @username VARCHAR(50),
    @password VARCHAR(255)
)
AS
BEGIN

    INSERT INTO Usuarios
    (
        username,
        password
    )
    VALUES
    (
        @username,
        @password
    )

END
GO

CREATE PROCEDURE ObtenerBitacora
(
    @fechaDesde DATETIME = NULL,
    @fechaHasta DATETIME = NULL
)
AS
BEGIN

    SELECT
        B.id_bitacora,
        U.username,
        B.mensaje,
        B.fecha,
	B.estado
    FROM Bitacora B
    INNER JOIN Usuarios U
        ON B.id_usuario = U.id_usuario
    WHERE
        (@fechaDesde IS NULL
            OR B.fecha >= @fechaDesde)
    AND
        (@fechaHasta IS NULL
            OR B.fecha <= @fechaHasta)
    ORDER BY B.fecha DESC

END
GO

CREATE PROCEDURE ActualizarUsuario
(
    @Id INT,
    @Username VARCHAR(100),
    @PasswordHash VARCHAR(200),
    @Email VARCHAR(100),
    @Telefono VARCHAR(30)
)
AS
BEGIN

    UPDATE Usuarios
    SET
        username = @Username,
        password = @PasswordHash,
	Email=@Email,
	Telefono=@Telefono
    WHERE id_usuario = @Id;

END
GO

CREATE TABLE AuditoriaUsuarios
(
    IdAuditoria INT IDENTITY(1,1) PRIMARY KEY,

    UsuarioId INT NOT NULL,

    Username VARCHAR(100) NOT NULL,

    PasswordHash VARCHAR(255) NOT NULL,

    Fecha DATETIME NOT NULL,

    ModificadoPor VARCHAR(100) NOT NULL,

    Accion VARCHAR(50) NOT NULL,

    CONSTRAINT FK_AuditoriaUsuarios_Usuarios
        FOREIGN KEY (UsuarioId)
        REFERENCES Usuarios(id_usuario)
);
GO

CREATE PROCEDURE [dbo].[Auditoria_ObtenerHistorial]
    @UsuarioId INT
AS
BEGIN
    SELECT * FROM AuditoriaUsuarios
    WHERE UsuarioId = @UsuarioId
    ORDER BY Fecha DESC

GO

CREATE PROCEDURE [dbo].[Auditoria_ObtenerPorId]
    @IdAuditoria INT
AS
BEGIN
    SELECT * FROM AuditoriaUsuarios WHERE IdAuditoria = @IdAuditoria
END
GO

CREATE TABLE Permiso

CREATE TABLE Permiso_Permiso
(
    Id_Permiso_Padre INT NOT NULL,
    Id_Permiso_Hijo INT NOT NULL,

    CONSTRAINT PK_Permiso_Permiso
        PRIMARY KEY (Id_Permiso_Padre, Id_Permiso_Hijo),

    CONSTRAINT FK_PP_Padre
        FOREIGN KEY (Id_Permiso_Padre)
        REFERENCES Permiso(Id),

    CONSTRAINT FK_PP_Hijo
        FOREIGN KEY (Id_Permiso_Hijo)
        REFERENCES Permiso(Id)
);
GO


CREATE TABLE Usuario_Permiso
(
    Id_Usuario INT NOT NULL,
    Id_Permiso INT NOT NULL,

    CONSTRAINT PK_Usuario_Permiso
        PRIMARY KEY (Id_Usuario, Id_Permiso),

    CONSTRAINT FK_UP_Usuario
        FOREIGN KEY (Id_Usuario)
        REFERENCES Usuario(id_usuario),

    CONSTRAINT FK_UP_Permiso
        FOREIGN KEY (Id_Permiso)
        REFERENCES Permiso(Id)
);
GO

CREATE PROCEDURE [dbo].[GetAllPatentes]
AS
BEGIN

    FROM Permiso
    WHERE Tipo IS NOT NULL;
END;
GO

AS
BEGIN

    FROM Permiso
    WHERE Tipo IS NULL;
END;
GO

CREATE PROCEDURE [dbo].[ObtenerPermisosUsuario]
    @usuarioId INT
AS
BEGIN
    SELECT p.id, p.nombre, p.Tipo
    FROM Usuario_Permiso up
    INNER JOIN permiso p ON up.id_permiso = p.id
    WHERE up.id_usuario = @usuarioId;
END
GO


CREATE PROCEDURE [dbo].[GetAllComponentes]
    @familiaId INT = NULL
AS
BEGIN
    WITH recursivo AS (
        SELECT sp2.id_permiso_padre, sp2.id_permiso_hijo
        FROM permiso_permiso sp2
        WHERE (@familiaId IS NULL AND sp2.id_permiso_padre IS NULL)
           OR (@familiaId IS NOT NULL AND sp2.id_permiso_padre = @familiaId)
        
        UNION ALL

        SELECT sp.id_permiso_padre, sp.id_permiso_hijo
        FROM permiso_permiso sp
        INNER JOIN recursivo r ON r.id_permiso_hijo = sp.id_permiso_padre
    )
    SELECT r.id_permiso_padre, r.id_permiso_hijo, p.id, p.nombre, p.Tipo
    FROM recursivo r
    INNER JOIN permiso p ON r.id_permiso_hijo = p.id;
END;
GO
